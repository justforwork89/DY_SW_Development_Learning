# LMS 실제 테이블 (참고)

> 원본: `03 프로젝트 산출물/LMS/LMS 테이블 정의서.md` (Oracle에서 내보낸 DDL). 보기 쉽게 저장 옵션(TABLESPACE, STORAGE 등)만 뺐고, FK가 먼저 만든 테이블을 가리키도록 만드는 순서만 바꿨어요. 컬럼·제약조건은 바꾸지 않았어요.
> 테이블 7개 · 컬럼 40개 · PK 7개 · UNIQUE 1개 · FK 9개 · 공통코드 13개

```sql
CREATE TABLE COMMON_CODE (
  GROUP_CD VARCHAR2(20),
  CODE VARCHAR2(10),
  CODE_NAME VARCHAR2(50),
  SORT_ORDER NUMBER,
  CONSTRAINT COMMON_CODE_PK PRIMARY KEY (GROUP_CD, CODE)
);

CREATE TABLE USERS (
  USER_ID NUMBER,
  LOGIN_ID VARCHAR2(50),
  PASSWORD VARCHAR2(100),
  USER_NAME VARCHAR2(50),
  ROLE_CD VARCHAR2(10),
  USE_YN CHAR(1),
  CONSTRAINT USERS_PK PRIMARY KEY (USER_ID),
  CONSTRAINT USERS_UK UNIQUE (LOGIN_ID)
);

CREATE TABLE COURSE (
  COURSE_ID NUMBER,
  COURSE_NAME VARCHAR2(100),
  INSTRUCTOR_ID NUMBER,
  START_DATE DATE,
  END_DATE DATE,
  CAPACITY NUMBER,
  STATUS_GROUP_CD VARCHAR2(20),
  STATUS_CD VARCHAR2(10),
  CONSTRAINT COURSE_PK PRIMARY KEY (COURSE_ID),
  CONSTRAINT FK_COURSE_USER FOREIGN KEY (INSTRUCTOR_ID) REFERENCES USERS (USER_ID),
  CONSTRAINT FK_COURSE_CODE FOREIGN KEY (STATUS_GROUP_CD, STATUS_CD) REFERENCES COMMON_CODE (GROUP_CD, CODE)
);

CREATE TABLE ENROLL (
  ENROLL_ID NUMBER,
  COURSE_ID NUMBER,
  USER_ID NUMBER,
  STATUS_GROUP_CD VARCHAR2(20),
  STATUS_CD VARCHAR2(10),
  ENROLL_DATE DATE,
  CONSTRAINT ENROLL_PK PRIMARY KEY (ENROLL_ID),
  CONSTRAINT FK_ENROLL_COURSE FOREIGN KEY (COURSE_ID) REFERENCES COURSE (COURSE_ID),
  CONSTRAINT FK_ENROLL_USER FOREIGN KEY (USER_ID) REFERENCES USERS (USER_ID),
  CONSTRAINT FK_ENROLL_CODE FOREIGN KEY (STATUS_GROUP_CD, STATUS_CD) REFERENCES COMMON_CODE (GROUP_CD, CODE)
);

CREATE TABLE ASSIGNMENT (
  ASSIGN_ID NUMBER,
  COURSE_ID NUMBER,
  TITLE VARCHAR2(200),
  CONTENT CLOB,
  DUE_DATE DATE,
  CONSTRAINT ASSIGNMENT_PK PRIMARY KEY (ASSIGN_ID),
  CONSTRAINT FK_ASSIGN_COURSE FOREIGN KEY (COURSE_ID) REFERENCES COURSE (COURSE_ID)
);

CREATE TABLE SUBMISSION (
  SUBMIT_ID NUMBER,
  ASSIGN_ID NUMBER,
  USER_ID NUMBER,
  CONTENT CLOB,
  SCORE NUMBER,
  STATUS_CD VARCHAR2(10),
  CONSTRAINT SUBMISSION_PK PRIMARY KEY (SUBMIT_ID),
  CONSTRAINT FK_SUBMIT_ASSIGN FOREIGN KEY (ASSIGN_ID) REFERENCES ASSIGNMENT (ASSIGN_ID),
  CONSTRAINT FK_SUBMIT_USER FOREIGN KEY (USER_ID) REFERENCES USERS (USER_ID)
);

CREATE TABLE FILE_DATA (
  FILE_ID NUMBER,
  SUBMIT_ID NUMBER,
  ORIGIN_NAME VARCHAR2(255),
  SAVE_PATH VARCHAR2(500),
  FILE_SIZE NUMBER,
  CONSTRAINT FILE_DATA_PK PRIMARY KEY (FILE_ID),
  CONSTRAINT FK_FILE_SUBMIT FOREIGN KEY (SUBMIT_ID) REFERENCES SUBMISSION (SUBMIT_ID)
);

```
